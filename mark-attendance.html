/** * mark-attendance.js * Lets the center team mark Present/Absent for every student in one of * their own center's ACTIVE batches (active status is read straight from * the Batches tab, auto-corrected server-side if the end date has passed), * for a chosen date, then bulk-submits it to the Attendance tab. Also lets * them update each student's Status (Active/Completed/Dropout) and * Placement-ready flag for the same batch. */ (function(){ const session = Layout.init(); if(!session) return; const els = { center: document.getElementById("f-center"), batch: document.getElementById("f-batch"), date: document.getElementById("f-date"), markAllPresent: document.getElementById("mark-all-present"), markAllAbsent: document.getElementById("mark-all-absent"), batchSummary: document.getElementById("batch-summary"), container: document.getElementById("attendance-container"), placementContainer: document.getElementById("placement-container") }; const todayIso = new Date().toLocaleDateString("en-CA"); els.date.value = todayIso; els.date.max = todayIso; // A single transient failure (brief network blip, or Google momentarily // rate-limiting a very active Apps Script deployment) shouldn't fail an // entire attendance/placement submission outright — retry once. async function postWithRetry(payload){ async function attempt(){ const res = await fetch(APP_CONFIG.APPS_SCRIPT_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(payload) }); if(!res.ok) throw new Error("API responded with status " + res.status); return res.json(); } try{ return await attempt(); }catch(err){ await new Promise(r => setTimeout(r, 1200)); return await attempt(); } } let latestData = null; let currentStudents = []; // attendance rows: [{studentId, studentName, status}] let currentPlacementStudents = []; // [{studentId, studentName, status, placementReady}] function populateCenterSelect(){ const allowed = Auth.allowedCenters(); els.center.innerHTML = allowed.map(c => ``).join(""); if(session.role === "center"){ els.center.value = session.center; els.center.disabled = true; } } function populateBatchSelect(){ if(!latestData) return; const center = els.center.value; // Only ACTIVE batches (per the Batches tab Status, auto-corrected if the // end date has already passed) — completed/closed batches don't take // new attendance here. const activeBatches = (latestData.batches||[]).filter(b => b.center === center && b.status === "Active"); els.batch.innerHTML = `` + activeBatches.map(b => ``).join(""); loadStudentsForBatch(); loadPlacementStudents(); renderBatchSummary(); } function renderBatchSummary(){ const batchId = els.batch.value; if(!batchId){ els.batchSummary.innerHTML = ""; return; } const batch = (latestData.batches||[]).find(b => b.batchId === batchId); if(!batch){ els.batchSummary.innerHTML = ""; return; } els.batchSummary.innerHTML = `
Enrolled${Utils.fmtInt(batch.enrolled)}
Today's Attendance %${Utils.fmtPct(batch.attendancePct)}
Overall Attendance %${Utils.fmtPct(batch.overallAttendancePct)}${batch.overallAttendancePct < 85 ? ' ⚠️' : ''}
Dropouts${Utils.fmtInt(batch.dropouts)}
`; } // ---------------- Attendance marking ---------------- function loadStudentsForBatch(){ const batchId = els.batch.value; if(!batchId){ currentStudents = []; renderTable(); return; } const students = (latestData.students||[]).filter(s => s.batchId === batchId && s.status !== "Dropout"); currentStudents = students.map(s => ({ studentId: s.studentId, studentName: s.studentName, status: "Present" })); renderTable(); } function renderTable(){ if(!els.batch.value){ els.container.innerHTML = `
Select a batch to load its students.
`; return; } if(!currentStudents.length){ els.container.innerHTML = `
No active students found in this batch.
`; return; } els.container.innerHTML = `
Marking attendance for ${Utils.escapeHtml(els.date.value)} — defaults to Present, tap to mark Absent.
${currentStudents.map((s,i) => ``).join("")}
Student IDNameStatus
${Utils.escapeHtml(s.studentId)} ${Utils.escapeHtml(s.studentName)}
`; Icons.hydrate(els.container); Utils.qsa(".status-toggle", els.container).forEach(btn => { btn.addEventListener("click", () => { const idx = Number(btn.dataset.idx); currentStudents[idx].status = currentStudents[idx].status === "Present" ? "Absent" : "Present"; btn.textContent = currentStudents[idx].status; btn.className = "status-toggle " + (currentStudents[idx].status === "Present" ? "present" : "absent"); }); }); document.getElementById("submit-attendance").addEventListener("click", submitAttendance); } async function submitAttendance(){ const btn = document.getElementById("submit-attendance"); const progressEl = document.getElementById("submit-progress"); btn.disabled = true; progressEl.textContent = "Submitting…"; const rows = currentStudents.map(s => ({ studentId: s.studentId, center: els.center.value, batchId: els.batch.value, date: els.date.value, status: s.status })); try{ const json = await postWithRetry({ action: "markAttendance", rows }); if(!json.ok) throw new Error(json.error || "Submit failed"); progressEl.textContent = `Saved — ${json.inserted} new, ${json.updated} updated.`; Utils.toast("Attendance saved", "ok"); Api.refreshNow(); }catch(err){ progressEl.textContent = `Failed: ${err.message}`; btn.disabled = false; } } // ---------------- Placement & Dropout update ---------------- const STUDENT_STATUSES = ["Active", "Completed", "Dropout"]; function loadPlacementStudents(){ const batchId = els.batch.value; if(!batchId){ currentPlacementStudents = []; renderPlacementTable(); return; } // Includes every student in the batch regardless of current status, so a // mistaken Dropout can be corrected back to Active/Completed here too. const students = (latestData.students||[]).filter(s => s.batchId === batchId); currentPlacementStudents = students.map(s => ({ studentId: s.studentId, studentName: s.studentName, status: s.status || "Active", placementReady: !!s.placementReady })); renderPlacementTable(); } function renderPlacementTable(){ if(!els.batch.value){ els.placementContainer.innerHTML = `
Select a batch to load its students.
`; return; } if(!currentPlacementStudents.length){ els.placementContainer.innerHTML = `
No students found in this batch.
`; return; } els.placementContainer.innerHTML = `
${currentPlacementStudents.map((s,i) => ``).join("")}
Student IDNameStatusPlacement Ready
${Utils.escapeHtml(s.studentId)} ${Utils.escapeHtml(s.studentName)}
`; Icons.hydrate(els.placementContainer); Utils.qsa(".placement-status-select", els.placementContainer).forEach(sel => { sel.addEventListener("change", () => { currentPlacementStudents[Number(sel.dataset.idx)].status = sel.value; }); }); Utils.qsa(".status-toggle", els.placementContainer).forEach(btn => { btn.addEventListener("click", () => { const idx = Number(btn.dataset.idx); currentPlacementStudents[idx].placementReady = !currentPlacementStudents[idx].placementReady; btn.textContent = currentPlacementStudents[idx].placementReady ? "Yes" : "No"; btn.className = "status-toggle " + (currentPlacementStudents[idx].placementReady ? "present" : "absent"); }); }); document.getElementById("submit-placement").addEventListener("click", submitPlacement); } async function submitPlacement(){ const btn = document.getElementById("submit-placement"); const progressEl = document.getElementById("placement-progress"); btn.disabled = true; progressEl.textContent = "Saving…"; const rows = currentPlacementStudents.map(s => ({ studentId: s.studentId, status: s.status, placementReady: s.placementReady })); try{ const json = await postWithRetry({ action: "updateStudents", rows }); if(!json.ok) throw new Error(json.error || "Save failed"); progressEl.textContent = `Saved — ${json.updated} student(s) updated${json.notFound ? `, ${json.notFound} not found` : ""}.`; Utils.toast("Student details updated", "ok"); Api.refreshNow(); }catch(err){ progressEl.textContent = `Failed: ${err.message}`; btn.disabled = false; } } els.center.addEventListener("change", populateBatchSelect); els.batch.addEventListener("change", () => { loadStudentsForBatch(); loadPlacementStudents(); renderBatchSummary(); }); els.date.addEventListener("change", () => { if(els.batch.value) loadStudentsForBatch(); }); els.markAllPresent.addEventListener("click", () => { currentStudents.forEach(s => s.status = "Present"); renderTable(); }); els.markAllAbsent.addEventListener("click", () => { currentStudents.forEach(s => s.status = "Absent"); renderTable(); }); populateCenterSelect(); Api.onData((data) => { if(!data) return; latestData = data; populateBatchSelect(); }); })();
